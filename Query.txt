TABLA DE TASAS DE CAMBIO

CREATE TABLE TAZA_DE_CAMBIO (
  fecha                   DATE           PRIMARY KEY,
  tasa_cambio_dollar      NUMBER(12,4)   NOT NULL,  -- cuántos quetzales por dólar
  tasa_cambio_quetzales   NUMBER(12,4)   NOT NULL   -- cuántos dólares por quetzal
);








PACKAGE CONVERSION MODENA


CREATE OR REPLACE PACKAGE PKG_CONVERTIDOR_MONEDA AS 
  PROCEDURE PROC_CONVERTIR_MONEDA(
    p_fecha            IN  DATE,
    p_monto            IN  NUMBER,
    p_tipo_conversion  IN  NUMBER,
    p_conversion       OUT NUMBER
  );
END PKG_CONVERTIDOR_MONEDA;
/

CREATE OR REPLACE PACKAGE BODY PKG_CONVERTIDOR_MONEDA AS

  PROCEDURE PROC_CONVERTIR_MONEDA(
    p_fecha            IN  DATE,
    p_monto            IN  NUMBER,
    p_tipo_conversion  IN  NUMBER,
    p_conversion       OUT NUMBER
  ) IS
    v_taza_de_cambio NUMBER(12,4);
  BEGIN
    IF p_tipo_conversion = 1 THEN
      SELECT tasa_cambio_dollar
        INTO v_taza_de_cambio
        FROM TAZA_DE_CAMBIO
       WHERE fecha = p_fecha;
      p_conversion := p_monto * v_taza_de_cambio;

    ELSIF p_tipo_conversion = 2 THEN
      SELECT tasa_cambio_quetzales
        INTO v_taza_de_cambio
        FROM TAZA_DE_CAMBIO
       WHERE fecha = p_fecha;
      p_conversion := p_monto * v_taza_de_cambio;

    ELSE

      p_conversion := 0;
    END IF;
  END PROC_CONVERTIR_MONEDA;

END PKG_CONVERTIDOR_MONEDA;
/
